from fastapi import FastAPI, HTTPException, Request from fastapi.staticfiles import StaticFiles from fastapi.responses import FileResponse from fastapi.middleware.cors import CORSMiddleware from pydantic import BaseModel, EmailStr import anthropic import json, uuid, os from datetime import datetime from pdf_generator import generate_pdf from prompt import build_prompt from stripe_handler import create_checkout_session, verify_session, construct_webhook_event app = FastAPI(title="Sterling Sovereign API", version="1.1.0") app.add_middleware(CORSMiddleware, allow_origins=["*"], allow_methods=["*"], allow_headers=["*"]) client = anthropic.Anthropic(api_key=os.environ.get("ANTHROPIC_API_KEY")) BASE_URL = os.environ.get("BASE_URL", "http://localhost:8000") app.mount("/static", StaticFiles(directory="static"), name="static") # ── Schemas ──────────────────────────────────────────────── class ProfileRequest(BaseModel): email: EmailStr full_name: str linkedin_url: str | None = None company: str | None = None sector: str | None = None annual_revenue: str | None = None years_experience: int | None = None founded_companies: int | None = 0 ambition_statement: str | None = None tier: str = "duke" class GenerateFromSession(BaseModel): session_id: str # ── Pages ────────────────────────────────────────────────── @app.get("/") def root(): return FileResponse("static/index.html") @app.get("/form") def form(): return FileResponse("static/form.html") @app.get("/success") def success_page(): return FileResponse("static/success.html") # ── Step 1 : create Stripe checkout ─────────────────────── @app.post("/create-checkout") async def create_checkout(req: ProfileRequest): metadata = { "email": req.email, "full_name": req.full_name, "linkedin_url": req.linkedin_url or "", "company": req.company or "", "sector": req.sector or "", "annual_revenue": req.annual_revenue or "", "years_experience": str(req.years_experience or ""), "founded_companies": str(req.founded_companies or 0), "ambition_statement": (req.ambition_statement or "")[:490], "tier": req.tier, } try: url = create_checkout_session( tier=req.tier, metadata=metadata, success_url=f"{BASE_URL}/success?session_id={{CHECKOUT_SESSION_ID}}", cancel_url=f"{BASE_URL}/form", ) return {"checkout_url": url} except Exception as e: raise HTTPException(status_code=500, detail=str(e)) # ── Step 2 : payment verified → generate profile ────────── @app.post("/generate-profile") async def generate_profile(req: GenerateFromSession): # Verify Stripe payment try: meta = verify_session(req.session_id) except ValueError as e: raise HTTPException(status_code=402, detail=str(e)) except Exception as e: raise HTTPException(status_code=500, detail=f"Stripe error: {e}") # Reconstruct profile data from Stripe metadata class R: email = meta.get("email", "") full_name = meta.get("full_name", "") linkedin_url = meta.get("linkedin_url") or None company = meta.get("company") or None sector = meta.get("sector") or None annual_revenue = meta.get("annual_revenue") or None years_experience = int(meta["years_experience"]) if meta.get("years_experience") else None founded_companies = int(meta.get("founded_companies", 0)) ambition_statement = meta.get("ambition_statement") or None tier = meta.get("tier", "duke") profile_id = str(uuid.uuid4())[:8].upper() # Call Claude prompt = build_prompt(R()) try: msg = client.messages.create( model="claude-opus-4-5", max_tokens=2048, messages=[{"role": "user", "content": prompt}] ) raw = msg.content[0].text except Exception as e: raise HTTPException(status_code=500, detail=f"Claude error: {e}") # Parse JSON from Claude try: clean = raw.strip() if clean.startswith("```"): clean = clean.split("```")[1] if clean.startswith("json"): clean = clean[4:] profile_data = json.loads(clean.strip()) except json.JSONDecodeError: raise HTTPException(status_code=500, detail="Failed to parse Claude response") profile_data["member_id"] = f"#{profile_id}" profile_data["tier"] = R.tier.capitalize() profile_data["full_name"] = R.full_name profile_data["generated_at"] = datetime.utcnow().strftime("%d %B %Y") # Generate PDF pdf_path = f"/tmp/sovereign_{profile_id}.pdf" generate_pdf(profile_data, pdf_path) return { "profile_id": profile_id, "pdf_url": f"/download/{profile_id}", "summary": { "prestige_score": profile_data.get("prestige_score"), "archetype": profile_data.get("archetype"), "rank": profile_data.get("rank"), "motto_latin": profile_data.get("motto_latin"), } } # ── PDF download ─────────────────────────────────────────── @app.get("/download/{profile_id}") def download_pdf(profile_id: str): path = f"/tmp/sovereign_{profile_id}.pdf" if not os.path.exists(path): raise HTTPException(status_code=404, detail="Profile not found") return FileResponse(path, media_type="application/pdf", filename=f"sterling_sovereign_{profile_id}.pdf") # ── Stripe webhook ───────────────────────────────────────── @app.post("/webhook/stripe") async def stripe_webhook(request: Request): payload = await request.body() sig_header = request.headers.get("stripe-signature", "") try: event = construct_webhook_event(payload, sig_header) except Exception: raise HTTPException(status_code=400, detail="Invalid webhook signature") if event["type"] == "checkout.session.completed": session = event["data"]["object"] print(f"[Stripe] Payment confirmed: {session.get('customer_email')} / {session.get('id')}") # TODO: send confirmation email return {"received": True} # ── Health ───────────────────────────────────────────────── @app.get("/health") def health(): return {"status": "sovereign", "timestamp": datetime.utcnow().isoformat()}
